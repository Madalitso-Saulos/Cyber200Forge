#include <stdio.h>
#include <string.h>

int check_serial(const char *input) {
    /* Simple XOR-based obfuscated comparison */
    const unsigned char key = 0x2A;
    const unsigned char target[] = {
        0x1c, 0x0e, 0x02, 0x0e, 0x1d, 0x02, 0x51, 0x02,
        0x18, 0x1d, 0x51, 0x02, 0x0c, 0x1d, 0x51, 0x1e,
        0x1a, 0x50, 0x1c, 0x02, 0x1d, 0x00, 0x0d, 0x1c,
        0x00, 0x1d, 0x1e, 0x53, 0x00
    };
    size_t target_len = sizeof(target);

    if (strlen(input) != target_len) {
        return 0;
    }
    for (size_t i = 0; i < target_len; i++) {
        if ((unsigned char)(input[i] ^ key) != target[i]) {
            return 0;
        }
    }
    return 1;
}

int main(int argc, char *argv[]) {
    if (argc != 2) {
        printf("Usage: %s <serial>\n", argv[0]);
        return 1;
    }

    if (check_serial(argv[1])) {
        printf("Access granted. Flag: FLAG{crackme_101_solved}\n");
    } else {
        printf("Access denied.\n");
    }

    return 0;
}
